
#!/bin/bash

# PROGRAM 6 - DOCKER BUILD PIPELINE

# 1. Install/check prerequisites
docker --version
node --version
npm --version
git --version

# 2. Generate package-lock.json
npm install

# 3. Build Docker image
docker build -t program-6:latest .

# 4. Check Docker image
docker images

# 5. Run Docker container
docker run -d -p 3000:3000 --name program6-container program-6:latest

# 6. Test application
curl http://localhost:3000

# 7. Stop and remove test container
docker rm -f program6-container

# 8. Docker Hub login
# Replace with your actual Docker Hub username
read -p "Docker Hub username: " DOCKERHUB_USERNAME
read -s -p "Docker Hub PAT: " DOCKERHUB_PAT
echo
printf '%s' "$DOCKERHUB_PAT" | docker login -u "$DOCKERHUB_USERNAME" --password-stdin
unset DOCKERHUB_PAT

# 9. Tag image for Docker Hub
docker tag program-6:latest "$DOCKERHUB_USERNAME/program-6:latest"

# 10. Push image to Docker Hub
docker push "$DOCKERHUB_USERNAME/program-6:latest"

# 11. GitHub Container Registry login
read -s -p "GitHub PAT classic: " GITHUB_PAT
echo
printf '%s' "$GITHUB_PAT" | docker login ghcr.io -u Spurthy-23 --password-stdin
unset GITHUB_PAT

# 12. Tag image for GHCR
docker tag program-6:latest ghcr.io/spurthy-23/program-6:latest

# 13. Push image to GHCR
docker push ghcr.io/spurthy-23/program-6:latest

# 14. Initialize Git repository
git init
git branch -M main

# 15. Add project files
git add .

# 16. Commit source code
git commit -m "Completed DevOps Program 6"

# 17. Connect GitHub repository
git remote add origin https://github.com/Spurthy-23/program-6.git

# 18. Push source code to GitHub
git push -u origin main

# PROGRAM 6 COMPLETED
